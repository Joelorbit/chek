curl -X POST https://api.chek.et/verify \
  -H "Content-Type: application/json" \
  -H "x-api-key: sk_live_9301e6a48..." \
  -d '{
    "input": "DHS78S7FQN"
  }'

          
          
const res = await fetch('https://api.chek.et/verify', {
  method: 'POST',
  headers: {
    'Content-Type': 'application/json',
    'x-api-key': process.env.CHEK_API_KEY
  },
  body: JSON.stringify({ input: 'DHS78S7FQN' })
});

const { success, provider, data } = await res.json();
if (success && data.amount >= 2500) {
  await fulfillCustomerOrder(orderId);

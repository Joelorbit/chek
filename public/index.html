curl -X POST http://localhost:3001/verify \
  -H "x-api-key: sk_live_YOUR_KEY_HERE" \
  -H "content-type: application/json" \
  -d '{"reference": "DHS78S7FQN"}'


        
import axios from 'axios';

const response = await axios.post('http://localhost:3001/verify', {
  reference: 'DHS78S7FQN'
}, {
  headers: { 'x-api-key': process.env.CHEK_API_KEY }
});

if (response.data.success) {
  console.log('Payment Confirmed:', response.data.data.payerName, response.data.data.settledAmount);
}


        
import requests

res = requests.post(
    'http://localhost:3001/verify',
    headers={'x-api-key': 'sk_live_YOUR_KEY_HERE'},
    json={'reference': 'DHS78S7FQN'}
)

data = res.json()
if data.get('success'):
    print(f"Payer: {data['data']['payerName']}, Amount: {data['data']['settledAmount']}")

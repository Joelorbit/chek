import { Telegraf } from 'telegraf';
import axios from 'axios';

const bot = new Telegraf(process.env.TELEGRAM_BOT_TOKEN);

bot.on('text', async (ctx) => {
  const text = ctx.message.text.trim();
  const res = await axios.post('http://localhost:3001/verify', {
    reference: text.includes(' ') ? undefined : text,
    receiptText: text.includes(' ') ? text : undefined
  }, { headers: { 'x-api-key': process.env.CHEK_KEY } });

  if (res.data.success) {
    await ctx.reply(`✅ Payment Verified: ${res.data.data.settledAmount} ETB!\nOrder Unlocked 🎉`);
  }
});
bot.launch();
